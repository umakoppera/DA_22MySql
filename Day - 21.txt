--------------------

-----------------------

offset function:
------------------

syntax of window function:
---------------------------
select columlist,functionname over() as aliasname from tablename;

in window function represented by over():
-------------------------------------------
partition by-----it is used to divided the data into group based on column
order by ---------to sort the data into a order based on column


select emp_id,emp_name,salary,rank(),over(order by salary desc) as rsalary
from employee;

select * from
(select emp_name,department,salary,rank() 
over(partition by department order by salary desc) as rsalary from employee)
where rsalary=1;

over(partition by department order by salary desc) 
as rsalary from employee) emp where rsalary<=2;
